{{- /* Returns the author's links elsewhere, as a slice of dicts: title short name, for the footer row label longer form where there is room -- the About page shows the address itself rather than the word "Email" url Driven by site.Params.author, so a key removed from the config drops its link rather than leaving a dead one. Used by the footer and by About. */ -}} {{- $a := site.Params.author -}} {{- $social := slice -}} {{- with $a.github }}{{ $social = $social | append (dict "title" "GitHub" "url" (printf "https://github.com/%s" .)) }}{{ end -}} {{- with $a.linkedin }}{{ $social = $social | append (dict "title" "LinkedIn" "url" (printf "https://linkedin.com/in/%s" .)) }}{{ end -}} {{- with $a.twitter }}{{ $social = $social | append (dict "title" "Twitter" "url" (printf "https://twitter.com/%s" .)) }}{{ end -}} {{- with index $a "500px" }}{{ $social = $social | append (dict "title" "500px" "url" (printf "https://500px.com/%s" .)) }}{{ end -}} {{- with $a.email }}{{ $social = $social | append (dict "title" "Email" "label" . "url" (printf "mailto:%s" .)) }}{{ end -}} {{- with site.Home.OutputFormats.Get "rss" }}{{ $social = $social | append (dict "title" "RSS" "label" "RSS feed" "url" .RelPermalink) }}{{ end -}} {{- return $social -}}
